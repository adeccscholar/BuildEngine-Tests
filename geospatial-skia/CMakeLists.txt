cmake_minimum_required(VERSION 3.25)
include("${CMAKE_CURRENT_SOURCE_DIR}/../bootstrap/PrepareToolchain.cmake")
project(GeoSpatialSkia LANGUAGES CXX)
include("${CMAKE_CURRENT_SOURCE_DIR}/../cmake/BuildEngineTests.cmake")
adecc_package_current_root(SKIA_PACKAGE skia)
adecc_package_current_root(TBB_PACKAGE onetbb)
adecc_package_current_root(OPENMP_PACKAGE llvm-openmp)
adecc_package_current_root(FASTFLOAT_PACKAGE fast_float)
list(PREPEND CMAKE_PREFIX_PATH "${SKIA_PACKAGE}")
find_package(Skia CONFIG REQUIRED)
find_package(TBB CONFIG REQUIRED COMPONENTS tbb
   PATHS "${TBB_PACKAGE}/lib/win64/${BUILDENGINE_CONFIGURATION}/cmake/TBB"
   NO_DEFAULT_PATH)
set(_omp "${OPENMP_PACKAGE}/lib/win64/${BUILDENGINE_CONFIGURATION}/libomp.lib")
if(NOT EXISTS "${_omp}" OR NOT EXISTS "${OPENMP_PACKAGE}/include/omp.h")
   message(FATAL_ERROR "Missing installed LLVM OpenMP shared runtime")
endif()
add_library(geo_openmp SHARED IMPORTED)
set_target_properties(geo_openmp PROPERTIES
   IMPORTED_IMPLIB "${_omp}" INTERFACE_INCLUDE_DIRECTORIES "${OPENMP_PACKAGE}/include")
add_library(geospatial_core OBJECT GeoData.cpp)
set_property(TARGET geospatial_core PROPERTY CXX_SCAN_FOR_MODULES OFF)
target_compile_features(geospatial_core PUBLIC cxx_std_23)
target_compile_options(geospatial_core PRIVATE -fopenmp)
target_include_directories(geospatial_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${FASTFLOAT_PACKAGE}/include")
target_link_libraries(geospatial_core PUBLIC TBB::tbb geo_openmp)
target_compile_definitions(geospatial_core PRIVATE __TBB_NO_IMPLICIT_LINKAGE=1)
add_executable(geospatial_benchmark benchmark.cpp)
set_property(TARGET geospatial_benchmark PROPERTY CXX_SCAN_FOR_MODULES OFF)
target_link_libraries(geospatial_benchmark PRIVATE geospatial_core)
add_executable(geospatial_viewer WIN32 viewer.cpp)
set_property(TARGET geospatial_viewer PROPERTY CXX_SCAN_FOR_MODULES OFF)
target_compile_features(geospatial_viewer PRIVATE cxx_std_23)
target_compile_definitions(geospatial_viewer PRIVATE
   UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(geospatial_viewer PRIVATE geospatial_core Skia::skia
   user32 gdi32 comdlg32 shell32)
target_link_options(geospatial_viewer PRIVATE "-Wl,--subsystem,windows")
foreach(_id IN ITEMS zlib bzip2 brotli libjpeg-turbo libpng harfbuzz freetype)
   adecc_package_current_root(_root "${_id}")
   list(APPEND _runtime_roots "${_root}")
endforeach()
list(APPEND _runtime_roots "${SKIA_PACKAGE}" "${TBB_PACKAGE}" "${OPENMP_PACKAGE}")
foreach(_target IN ITEMS geospatial_benchmark geospatial_viewer)
   adecc_test_install(${_target}
      PACKAGE_ROOTS ${_runtime_roots}
      TOOLCHAIN_DLLS libc++-370.dll libunwind-370.dll)
endforeach()
