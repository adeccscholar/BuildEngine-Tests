cmake_minimum_required(VERSION 3.25)
include("${CMAKE_CURRENT_SOURCE_DIR}/../bootstrap/PrepareToolchain.cmake")
project(BuildEngineOpenXlsxGrid LANGUAGES CXX)
include("${CMAKE_CURRENT_SOURCE_DIR}/../cmake/BuildEngineTests.cmake")
foreach(_id IN ITEMS openxlsx pugixml libzip boost zlib)
 string(TOUPPER "${_id}" _key)
 adecc_package_current_root(${_key}_PACKAGE "${_id}")
endforeach()
set(OpenXLSX_DIR "${OPENXLSX_PACKAGE}/lib/win64/${BUILDENGINE_CONFIGURATION}/cmake/OpenXLSX")
find_package(OpenXLSX CONFIG REQUIRED PATHS "${OpenXLSX_DIR}" NO_DEFAULT_PATH)
add_executable(adecc_openxlsx_grid_test main.cpp)
set_target_properties(adecc_openxlsx_grid_test PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF CXX_SCAN_FOR_MODULES OFF)
target_compile_definitions(adecc_openxlsx_grid_test PRIVATE CHRONO_WORK_AROUND CHRONO_MISSED_ZONEDB)
target_compile_options(adecc_openxlsx_grid_test PRIVATE "-fborland-exec-code-page=65001")
adecc_test_use_core(adecc_openxlsx_grid_test openxlsx)
target_link_libraries(adecc_openxlsx_grid_test PRIVATE OpenXLSX::OpenXLSX)
adecc_test_install(adecc_openxlsx_grid_test RUNTIME_CLOSURE PACKAGE_ROOTS
 "${OPENXLSX_PACKAGE}" "${PUGIXML_PACKAGE}" "${LIBZIP_PACKAGE}" "${BOOST_PACKAGE}" "${ZLIB_PACKAGE}"
 TOOLCHAIN_DLLS libc++-370.dll libunwind-370.dll)
enable_testing()
add_test(NAME OpenXlsxGridRoundtrip COMMAND "$<TARGET_FILE:adecc_openxlsx_grid_test>")
