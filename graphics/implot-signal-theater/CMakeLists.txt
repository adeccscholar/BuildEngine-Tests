cmake_minimum_required(VERSION 3.25)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../bootstrap/PrepareToolchain.cmake")
project(implot_signal_theater LANGUAGES CXX)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/BuildEngineTests.cmake")
include(CTest)
adecc_package_current_root(IMGUI_PACKAGE imgui)
adecc_package_current_root(IMPLOT_PACKAGE implot)
adecc_package_current_root(SDL2_PACKAGE sdl2)
list(PREPEND CMAKE_PREFIX_PATH "${SDL2_PACKAGE}")
find_package(SDL2 CONFIG REQUIRED PATHS "${SDL2_PACKAGE}" NO_DEFAULT_PATH)
if(BUILDENGINE_CONFIGURATION STREQUAL "Debug")
   set(_debug d)
else()
   set(_debug "")
endif()
set(_imgui_import "${IMGUI_PACKAGE}/lib/win64/${BUILDENGINE_CONFIGURATION}/libimgui${_debug}.lib")
set(_implot_import "${IMPLOT_PACKAGE}/lib/win64/${BUILDENGINE_CONFIGURATION}/libimplot${_debug}.lib")
foreach(_import IN ITEMS "${_imgui_import}" "${_implot_import}")
   if(NOT EXISTS "${_import}")
      message(FATAL_ERROR "Missing managed DLL import: ${_import}")
   endif()
endforeach()
add_executable(implot-signal-theater WIN32 main.cpp)
set_target_properties(implot-signal-theater PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF CXX_SCAN_FOR_MODULES OFF)
target_compile_definitions(implot-signal-theater PRIVATE SDL_MAIN_HANDLED IMGUI_API=__declspec(dllimport) IMPLOT_API=__declspec(dllimport) WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(implot-signal-theater PRIVATE "${IMGUI_PACKAGE}/include" "${IMPLOT_PACKAGE}/include" "${SDL2_PACKAGE}/include/SDL2")
target_link_libraries(implot-signal-theater PRIVATE "${_implot_import}" "${_imgui_import}" SDL2::SDL2)
if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
   target_link_options(implot-signal-theater PRIVATE "-Wl,--subsystem,windows")
endif()
adecc_test_install(implot-signal-theater RUNTIME_CLOSURE PACKAGE_ROOTS "${IMGUI_PACKAGE}" "${IMPLOT_PACKAGE}" "${SDL2_PACKAGE}" TOOLCHAIN_DLLS libc++-370.dll libunwind-370.dll)
if(BUILD_TESTING)
   add_test(NAME implot-signal-theater-verify COMMAND "$<TARGET_FILE:implot-signal-theater>" --verify)
   set_tests_properties(implot-signal-theater-verify PROPERTIES LABELS "native;showcase;offline")
endif()
