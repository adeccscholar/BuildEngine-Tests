cmake_minimum_required(VERSION 3.25)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../bootstrap/PrepareToolchain.cmake")
project(implot_signal_theater LANGUAGES CXX)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/BuildEngineTests.cmake")
include(CTest)
adecc_package_current_root(PACKAGE_0 imgui)
adecc_package_current_root(PACKAGE_1 implot)
if(BUILDENGINE_CONFIGURATION STREQUAL "Debug")
   set(_debug d)
else()
   set(_debug "")
endif()
set(_lib_0 "${PACKAGE_0}/lib/win64/${BUILDENGINE_CONFIGURATION}/libimgui${_debug}.lib")
if(NOT EXISTS "${_lib_0}")
   message(FATAL_ERROR "Missing native import library: ${_lib_0}")
endif()
set(_lib_1 "${PACKAGE_1}/lib/win64/${BUILDENGINE_CONFIGURATION}/libimplot${_debug}.lib")
if(NOT EXISTS "${_lib_1}")
   message(FATAL_ERROR "Missing native import library: ${_lib_1}")
endif()
add_executable(implot-signal-theater main.cpp)
set_target_properties(implot-signal-theater PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF CXX_SCAN_FOR_MODULES OFF)
target_include_directories(implot-signal-theater PRIVATE "${PACKAGE_0}/include" "${PACKAGE_1}/include")
target_link_libraries(implot-signal-theater PRIVATE "${_lib_0}" "${_lib_1}")
adecc_test_install(implot-signal-theater RUNTIME_CLOSURE PACKAGE_ROOTS "${PACKAGE_0}" "${PACKAGE_1}" TOOLCHAIN_DLLS libc++-370.dll libunwind-370.dll)
if(BUILD_TESTING)
   add_test(NAME implot-signal-theater-verify COMMAND "$<TARGET_FILE:implot-signal-theater>" --verify)
   set_tests_properties(implot-signal-theater-verify PROPERTIES LABELS "native;showcase;offline")
endif()
