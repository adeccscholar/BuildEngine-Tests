cmake_minimum_required(VERSION 3.31)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../bootstrap/PrepareToolchain.cmake")
project(acoustic_orbitarium LANGUAGES CXX)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/BuildEngineTests.cmake")
include(CTest)
adecc_package_current_root(OPENAL_PACKAGE openal-soft)
adecc_package_current_root(IMGUI_PACKAGE imgui)
adecc_package_current_root(SDL2_PACKAGE sdl2)
find_package(SDL2 CONFIG REQUIRED PATHS "${SDL2_PACKAGE}" NO_DEFAULT_PATH)
find_library(OPENAL_IMPORT NAMES OpenAL32 libOpenAL32 OpenAL libopenal PATHS "${OPENAL_PACKAGE}/lib/win64/${BUILDENGINE_CONFIGURATION}" NO_DEFAULT_PATH REQUIRED)
if(BUILDENGINE_CONFIGURATION STREQUAL "Debug")
   set(_suffix d)
else()
   set(_suffix "")
endif()
set(_imgui "${IMGUI_PACKAGE}/lib/win64/${BUILDENGINE_CONFIGURATION}/libimgui${_suffix}.lib")
if(NOT EXISTS "${_imgui}")
   message(FATAL_ERROR "Managed ImGui import missing: ${_imgui}")
endif()
add_executable(acoustic-orbitarium WIN32 main.cpp)
set_target_properties(acoustic-orbitarium PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF CXX_SCAN_FOR_MODULES OFF)
target_compile_definitions(acoustic-orbitarium PRIVATE SDL_MAIN_HANDLED WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(acoustic-orbitarium PRIVATE "${OPENAL_PACKAGE}/include" "${IMGUI_PACKAGE}/include" "${SDL2_PACKAGE}/include/SDL2")
target_link_libraries(acoustic-orbitarium PRIVATE "${OPENAL_IMPORT}" "${_imgui}" SDL2::SDL2)
target_link_options(acoustic-orbitarium PRIVATE "-Wl,--subsystem,windows")
adecc_test_install(acoustic-orbitarium RUNTIME_CLOSURE PACKAGE_ROOTS "${OPENAL_PACKAGE}" "${IMGUI_PACKAGE}" "${SDL2_PACKAGE}" TOOLCHAIN_DLLS libc++-370.dll libunwind-370.dll)
if(BUILD_TESTING)
   add_test(NAME AcousticOrbitariumVerify COMMAND "$<TARGET_FILE:acoustic-orbitarium>" --verify)
   set_tests_properties(AcousticOrbitariumVerify PROPERTIES LABELS "native;openal;showcase" TIMEOUT 60)
endif()
