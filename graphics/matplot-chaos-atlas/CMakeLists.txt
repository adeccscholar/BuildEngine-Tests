cmake_minimum_required(VERSION 3.25)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../bootstrap/PrepareToolchain.cmake")
project(matplot_chaos_atlas LANGUAGES CXX)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/BuildEngineTests.cmake")
include(CTest)
adecc_package_current_root(PACKAGE_0 matplotplusplus)
if(BUILDENGINE_CONFIGURATION STREQUAL "Debug")
   set(_debug d)
else()
   set(_debug "")
endif()
find_library(_lib_0 NAMES matplot libmatplot PATHS "${PACKAGE_0}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(_lib_1 NAMES nodesoup libnodesoup PATHS "${PACKAGE_0}/lib/Matplot++" NO_DEFAULT_PATH REQUIRED)
add_executable(matplot-chaos-atlas main.cpp)
set_target_properties(matplot-chaos-atlas PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF CXX_SCAN_FOR_MODULES OFF)
target_include_directories(matplot-chaos-atlas PRIVATE "${PACKAGE_0}/include")
target_link_libraries(matplot-chaos-atlas PRIVATE "${_lib_0}" "${_lib_1}")
adecc_test_install(matplot-chaos-atlas RUNTIME_CLOSURE PACKAGE_ROOTS "${PACKAGE_0}" TOOLCHAIN_DLLS libc++-370.dll libunwind-370.dll)
if(BUILD_TESTING)
   add_test(NAME matplot-chaos-atlas-verify COMMAND "$<TARGET_FILE:matplot-chaos-atlas>" --verify)
   set_tests_properties(matplot-chaos-atlas-verify PROPERTIES LABELS "native;showcase;offline")
endif()
