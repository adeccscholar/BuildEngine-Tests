cmake_minimum_required(VERSION 3.25)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../bootstrap/PrepareToolchain.cmake")
project(matplot_chaos_atlas LANGUAGES CXX)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/BuildEngineTests.cmake")
include(CTest)
adecc_package_current_root(MATPLOT_PACKAGE matplotplusplus)
adecc_package_current_root(SDL2_PACKAGE sdl2)
list(PREPEND CMAKE_PREFIX_PATH "${SDL2_PACKAGE}")
find_package(SDL2 CONFIG REQUIRED PATHS "${SDL2_PACKAGE}" NO_DEFAULT_PATH)
find_library(MATPLOT_IMPORT NAMES matplot libmatplot PATHS "${MATPLOT_PACKAGE}/lib" NO_DEFAULT_PATH REQUIRED)
find_library(NODESOUP_IMPORT NAMES nodesoup libnodesoup PATHS "${MATPLOT_PACKAGE}/lib/Matplot++" NO_DEFAULT_PATH REQUIRED)
add_executable(matplot-chaos-atlas WIN32 main.cpp)
set_target_properties(matplot-chaos-atlas PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF CXX_SCAN_FOR_MODULES OFF)
target_compile_definitions(matplot-chaos-atlas PRIVATE SDL_MAIN_HANDLED WIN32_LEAN_AND_MEAN NOMINMAX)
if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
   target_link_options(matplot-chaos-atlas PRIVATE "-Wl,--subsystem,windows")
endif()
target_include_directories(matplot-chaos-atlas PRIVATE "${MATPLOT_PACKAGE}/include" "${SDL2_PACKAGE}/include/SDL2")
target_link_libraries(matplot-chaos-atlas PRIVATE "${MATPLOT_IMPORT}" "${NODESOUP_IMPORT}" SDL2::SDL2)
adecc_test_install(matplot-chaos-atlas RUNTIME_CLOSURE PACKAGE_ROOTS "${MATPLOT_PACKAGE}" "${SDL2_PACKAGE}" TOOLCHAIN_DLLS libc++-370.dll libunwind-370.dll)
if(BUILD_TESTING)
   add_test(NAME matplot-chaos-atlas-verify COMMAND "$<TARGET_FILE:matplot-chaos-atlas>" --verify)
   set_tests_properties(matplot-chaos-atlas-verify PROPERTIES LABELS "native;showcase;offline")
endif()
