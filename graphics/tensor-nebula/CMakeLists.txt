cmake_minimum_required(VERSION 3.31)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../bootstrap/PrepareToolchain.cmake")
project(tensor_nebula LANGUAGES CXX)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/BuildEngineTests.cmake")
include(CTest)
adecc_package_current_root(XTENSOR_PACKAGE xtensor)
adecc_package_current_root(XTL_PACKAGE xtl)
adecc_package_current_root(XSIMD_PACKAGE xsimd)
adecc_package_current_root(TBB_PACKAGE onetbb)
adecc_package_current_root(IMGUI_PACKAGE imgui)
adecc_package_current_root(SDL2_PACKAGE sdl2)
adecc_package_current_root(OPENGL_PACKAGE opengl)
adecc_package_current_root(GLEW_PACKAGE glew)
list(PREPEND CMAKE_PREFIX_PATH "${SDL2_PACKAGE}" "${OPENGL_PACKAGE}" "${GLEW_PACKAGE}")
set(OpenGL_DIR "${OPENGL_PACKAGE}/lib/win64/${BUILDENGINE_CONFIGURATION}/cmake/OpenGL")
set(GLEW_DIR "${GLEW_PACKAGE}/lib/win64/${BUILDENGINE_CONFIGURATION}/cmake/GLEW")
find_package(OpenGL CONFIG REQUIRED PATHS "${OpenGL_DIR}" NO_DEFAULT_PATH)
find_package(GLEW CONFIG REQUIRED PATHS "${GLEW_DIR}" NO_DEFAULT_PATH)
if(NOT EXISTS "${OpenGL_RUNTIME_DLL}" OR NOT EXISTS "${OpenGL_GALLIUM_DLL}")
   message(FATAL_ERROR "Managed Mesa OpenGL and Gallium WGL runtimes are required")
endif()
set(TBB_DIR "${TBB_PACKAGE}/lib/win64/${BUILDENGINE_CONFIGURATION}/cmake/TBB")
find_package(TBB 2023.2 CONFIG REQUIRED PATHS "${TBB_DIR}" NO_DEFAULT_PATH)
find_package(SDL2 CONFIG REQUIRED PATHS "${SDL2_PACKAGE}" NO_DEFAULT_PATH)
if(BUILDENGINE_CONFIGURATION STREQUAL "Debug")
   set(_suffix d)
else()
   set(_suffix "")
endif()
set(_imgui "${IMGUI_PACKAGE}/lib/win64/${BUILDENGINE_CONFIGURATION}/libimgui${_suffix}.lib")
set(_imgui_backend "${IMGUI_PACKAGE}/lib/win64/${BUILDENGINE_CONFIGURATION}/libimgui_sdl2_opengl3${_suffix}.lib")
foreach(_lib IN ITEMS "${_imgui}" "${_imgui_backend}")
   if(NOT EXISTS "${_lib}")
      message(FATAL_ERROR "Managed ImGui native DLL import missing: ${_lib}")
   endif()
endforeach()
add_executable(tensor-nebula WIN32 main.cpp)
set_target_properties(tensor-nebula PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF CXX_SCAN_FOR_MODULES OFF)
target_compile_definitions(tensor-nebula PRIVATE SDL_MAIN_HANDLED WIN32_LEAN_AND_MEAN NOMINMAX XTENSOR_USE_XSIMD ADECC_MESA_OPENGL_DLL="${OpenGL_RUNTIME_DLL}")
target_include_directories(tensor-nebula PRIVATE "${XTENSOR_PACKAGE}/include" "${XTL_PACKAGE}/include" "${XSIMD_PACKAGE}/include" "${GLEW_PACKAGE}/include" "${OPENGL_PACKAGE}/include" "${IMGUI_PACKAGE}/include" "${SDL2_PACKAGE}/include/SDL2")
target_link_libraries(tensor-nebula PRIVATE "${_imgui}" "${_imgui_backend}" SDL2::SDL2 OpenGL::OpenGL GLEW::GLEW TBB::tbb)
target_link_options(tensor-nebula PRIVATE "-Wl,--subsystem,windows")
adecc_test_install(tensor-nebula RUNTIME_CLOSURE
   PACKAGE_ROOTS "${XTENSOR_PACKAGE}" "${XTL_PACKAGE}" "${XSIMD_PACKAGE}" "${TBB_PACKAGE}"
      "${OPENGL_PACKAGE}" "${GLEW_PACKAGE}" "${IMGUI_PACKAGE}" "${SDL2_PACKAGE}"
   TOOLCHAIN_DLLS libc++-370.dll libunwind-370.dll
   DYNAMIC_DLLS opengl32.dll libgallium_wgl.dll)
if(BUILD_TESTING)
   add_test(NAME TensorNebulaVerify COMMAND "$<TARGET_FILE:tensor-nebula>" --verify)
   set_tests_properties(TensorNebulaVerify PROPERTIES LABELS "native;xtensor;xsimd;xtl;onetbb;headless;showcase" TIMEOUT 60)
   option(TENSOR_NEBULA_ENABLE_GL_CTEST "Run native Mesa visual context checks on desktop hosts" OFF)
   if(TENSOR_NEBULA_ENABLE_GL_CTEST)
      add_test(NAME TensorNebulaMesaOpenGL COMMAND "$<TARGET_FILE:tensor-nebula>" --verify-gl)
      set_tests_properties(TensorNebulaMesaOpenGL PROPERTIES LABELS "native;mesa;glew;imgui;gpu;showcase" TIMEOUT 60)
   endif()
endif()
