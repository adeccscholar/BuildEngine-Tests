cmake_minimum_required(VERSION 3.25)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../bootstrap/PrepareToolchain.cmake")
project(imgui_neon_lab LANGUAGES CXX)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/BuildEngineTests.cmake")
include(CTest)
adecc_package_current_root(IMGUI_PACKAGE imgui)
adecc_package_current_root(SDL2_PACKAGE sdl2)
list(PREPEND CMAKE_PREFIX_PATH "${SDL2_PACKAGE}")
find_package(SDL2 CONFIG REQUIRED PATHS "${SDL2_PACKAGE}" NO_DEFAULT_PATH)
if(BUILDENGINE_CONFIGURATION STREQUAL "Debug")
   set(_debug "d")
else()
   set(_debug "")
endif()
set(_imgui_import "${IMGUI_PACKAGE}/lib/win64/${BUILDENGINE_CONFIGURATION}/libimgui${_debug}.lib")
if(NOT EXISTS "${_imgui_import}")
   message(FATAL_ERROR "Missing managed ImGui import: ${_imgui_import}")
endif()
add_executable(imgui-neon-lab WIN32 main.cpp)
set_target_properties(imgui-neon-lab PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF CXX_SCAN_FOR_MODULES OFF)
target_compile_definitions(imgui-neon-lab PRIVATE SDL_MAIN_HANDLED IMGUI_API=__declspec(dllimport) WIN32_LEAN_AND_MEAN NOMINMAX)
target_include_directories(imgui-neon-lab PRIVATE "${IMGUI_PACKAGE}/include" "${SDL2_PACKAGE}/include/SDL2")
target_link_libraries(imgui-neon-lab PRIVATE "${_imgui_import}" SDL2::SDL2)
if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
   target_link_options(imgui-neon-lab PRIVATE "-Wl,--subsystem,windows")
endif()
adecc_test_install(imgui-neon-lab RUNTIME_CLOSURE PACKAGE_ROOTS "${IMGUI_PACKAGE}" "${SDL2_PACKAGE}" TOOLCHAIN_DLLS libc++-370.dll libunwind-370.dll)
if(BUILD_TESTING)
   add_test(NAME imgui-neon-lab-verify COMMAND "$<TARGET_FILE:imgui-neon-lab>" --verify)
   set_tests_properties(imgui-neon-lab-verify PROPERTIES LABELS "native;showcase;offline")
endif()
