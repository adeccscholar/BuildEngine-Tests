cmake_minimum_required(VERSION 3.25)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../bootstrap/PrepareToolchain.cmake")
project(imgui_neon_lab LANGUAGES CXX)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/BuildEngineTests.cmake")
include(CTest)
adecc_package_current_root(PACKAGE_0 imgui)
if(BUILDENGINE_CONFIGURATION STREQUAL "Debug")
   set(_debug d)
else()
   set(_debug "")
endif()
set(_lib_0 "${PACKAGE_0}/lib/win64/${BUILDENGINE_CONFIGURATION}/libimgui${_debug}.lib")
if(NOT EXISTS "${_lib_0}")
   message(FATAL_ERROR "Missing native import library: ${_lib_0}")
endif()
add_executable(imgui-neon-lab main.cpp)
set_target_properties(imgui-neon-lab PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF CXX_SCAN_FOR_MODULES OFF)
target_include_directories(imgui-neon-lab PRIVATE "${PACKAGE_0}/include")
target_link_libraries(imgui-neon-lab PRIVATE "${_lib_0}")
adecc_test_install(imgui-neon-lab RUNTIME_CLOSURE PACKAGE_ROOTS "${PACKAGE_0}" TOOLCHAIN_DLLS libc++-370.dll libunwind-370.dll)
if(BUILD_TESTING)
   add_test(NAME imgui-neon-lab-verify COMMAND "$<TARGET_FILE:imgui-neon-lab>" --verify)
   set_tests_properties(imgui-neon-lab-verify PROPERTIES LABELS "native;showcase;offline")
endif()
