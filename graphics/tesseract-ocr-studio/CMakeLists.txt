cmake_minimum_required(VERSION 3.25)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../bootstrap/PrepareToolchain.cmake")
project(tesseract_ocr_studio LANGUAGES CXX)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/BuildEngineTests.cmake")
include(CTest)

adecc_package_current_root(OPENCV_PACKAGE opencv)
adecc_package_current_root(TESSERACT_PACKAGE tesseract)
adecc_package_current_root(LEPTONICA_PACKAGE leptonica)
adecc_package_current_root(IMGUI_PACKAGE imgui)
adecc_package_current_root(SDL2_PACKAGE sdl2)
adecc_package_current_root(TBB_PACKAGE onetbb)
adecc_package_current_root(OMP_PACKAGE llvm-openmp)
adecc_package_current_root(ZLIB_PACKAGE zlib)
adecc_package_current_root(PNG_PACKAGE libpng)
adecc_package_current_root(JPEG_PACKAGE libjpeg-turbo)
adecc_package_current_root(TIFF_PACKAGE libtiff)
set(OpenCV_DIR "${OPENCV_PACKAGE}/lib/win64/${BUILDENGINE_CONFIGURATION}/cmake/opencv5")
set(Tesseract_DIR "${TESSERACT_PACKAGE}/lib/win64/${BUILDENGINE_CONFIGURATION}/cmake/tesseract")
set(Leptonica_DIR "${LEPTONICA_PACKAGE}/lib/win64/${BUILDENGINE_CONFIGURATION}/cmake/leptonica")
find_package(OpenCV 5.1 CONFIG REQUIRED COMPONENTS core imgproc imgcodecs videoio PATHS "${OpenCV_DIR}" NO_DEFAULT_PATH)
find_package(Leptonica 1.87 CONFIG REQUIRED PATHS "${Leptonica_DIR}" NO_DEFAULT_PATH)
find_package(Tesseract 5.5 CONFIG REQUIRED PATHS "${Tesseract_DIR}" NO_DEFAULT_PATH)
find_package(SDL2 CONFIG REQUIRED PATHS "${SDL2_PACKAGE}" NO_DEFAULT_PATH)
set(_tessdata "${TESSERACT_PACKAGE}/share/tessdata")
if(NOT EXISTS "${_tessdata}/eng.traineddata")
   message(FATAL_ERROR "Managed English OCR model missing: ${_tessdata}/eng.traineddata")
endif()
if(BUILDENGINE_CONFIGURATION STREQUAL "Debug")
   set(_imgui_suffix d)
else()
   set(_imgui_suffix "")
endif()
set(_imgui_import "${IMGUI_PACKAGE}/lib/win64/${BUILDENGINE_CONFIGURATION}/libimgui${_imgui_suffix}.lib")
if(NOT EXISTS "${_imgui_import}")
   message(FATAL_ERROR "Managed Dear ImGui import missing: ${_imgui_import}")
endif()
add_executable(tesseract-ocr-studio WIN32 main.cpp)
set_target_properties(tesseract-ocr-studio PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF CXX_SCAN_FOR_MODULES OFF)
target_compile_definitions(tesseract-ocr-studio PRIVATE
   SDL_MAIN_HANDLED WIN32_LEAN_AND_MEAN NOMINMAX
   ADECC_TESSDATA_PATH="${_tessdata}")
target_include_directories(tesseract-ocr-studio PRIVATE
   ${OpenCV_INCLUDE_DIRS} ${Tesseract_INCLUDE_DIRS} ${Leptonica_INCLUDE_DIRS}
   "${IMGUI_PACKAGE}/include" "${SDL2_PACKAGE}/include/SDL2")
target_link_libraries(tesseract-ocr-studio PRIVATE
   ${OpenCV_LIBS} Tesseract::libtesseract "${_imgui_import}" SDL2::SDL2 ole32 oleaut32 strmiids)
if(WIN32 AND CMAKE_CXX_COMPILER_ID STREQUAL "Clang")
   target_link_options(tesseract-ocr-studio PRIVATE "-Wl,--subsystem,windows")
endif()
adecc_test_install(tesseract-ocr-studio RUNTIME_CLOSURE
   PACKAGE_ROOTS "${OPENCV_PACKAGE}" "${TESSERACT_PACKAGE}" "${LEPTONICA_PACKAGE}"
      "${IMGUI_PACKAGE}" "${SDL2_PACKAGE}" "${TBB_PACKAGE}" "${OMP_PACKAGE}"
      "${ZLIB_PACKAGE}" "${PNG_PACKAGE}" "${JPEG_PACKAGE}" "${TIFF_PACKAGE}"
   TOOLCHAIN_DLLS libc++-370.dll libunwind-370.dll)
if(BUILD_TESTING)
   add_test(NAME TesseractOCRStudioNative COMMAND "$<TARGET_FILE:tesseract-ocr-studio>" --verify)
   set_tests_properties(TesseractOCRStudioNative PROPERTIES LABELS "native;tesseract;leptonica;opencv;ocr;showcase" TIMEOUT 120)
endif()
