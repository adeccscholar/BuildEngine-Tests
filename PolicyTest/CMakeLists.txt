cmake_minimum_required(VERSION 3.25)
include("${CMAKE_CURRENT_SOURCE_DIR}/../bootstrap/PrepareToolchain.cmake")
project(BuildEnginePolicyTests LANGUAGES CXX)
include("${CMAKE_CURRENT_SOURCE_DIR}/../cmake/BuildEngineTests.cmake")
add_executable(adecc_policy_test test_policy.cpp)
set_target_properties(adecc_policy_test PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF CXX_SCAN_FOR_MODULES OFF)
target_compile_definitions(adecc_policy_test PRIVATE CHRONO_WORK_AROUND CHRONO_MISSED_ZONEDB)
target_compile_options(adecc_policy_test PRIVATE "-fborland-exec-code-page=65001")
adecc_test_use_core(adecc_policy_test domain)
adecc_test_install(adecc_policy_test RUNTIME_CLOSURE TOOLCHAIN_DLLS libc++-370.dll libunwind-370.dll)
enable_testing()
add_test(NAME PolicyAcceptance COMMAND "$<TARGET_FILE:adecc_policy_test>")
set_tests_properties(PolicyAcceptance PROPERTIES LABELS "adecc-core-lib;policy;negative")
