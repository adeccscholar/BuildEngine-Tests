cmake_minimum_required(VERSION 3.25)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../bootstrap/PrepareToolchain.cmake")
project(BuildEngineMssqlTests LANGUAGES CXX)
include("${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/BuildEngineTests.cmake")
adecc_package_current_root(OPENXLSX_PACKAGE openxlsx)
adecc_package_current_root(PUGIXML_PACKAGE pugixml)
adecc_package_current_root(LIBZIP_PACKAGE libzip)
adecc_package_current_root(BOOST_PACKAGE boost)
foreach(_pkg IN ITEMS "${OPENXLSX_PACKAGE}" "${PUGIXML_PACKAGE}" "${LIBZIP_PACKAGE}" "${BOOST_PACKAGE}")
 list(PREPEND CMAKE_PREFIX_PATH "${_pkg}")
endforeach()
set(OpenXLSX_DIR "${OPENXLSX_PACKAGE}/lib/win64/${BUILDENGINE_CONFIGURATION}/cmake/OpenXLSX")
find_package(OpenXLSX CONFIG REQUIRED PATHS "${OpenXLSX_DIR}" NO_DEFAULT_PATH)
foreach(_target IN ITEMS adecc_mssql_odbc_test adecc_mssql_csv_pipeline)
 if(_target STREQUAL "adecc_mssql_odbc_test")
  add_executable(${_target} test_mssql_odbc_database.cpp)
 else()
  add_executable(${_target} csv_booking_pipeline.cpp)
 endif()
 set_target_properties(${_target} PROPERTIES CXX_STANDARD 23 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF CXX_SCAN_FOR_MODULES OFF)
 target_compile_definitions(${_target} PRIVATE CHRONO_WORK_AROUND CHRONO_MISSED_ZONEDB)
 target_compile_options(${_target} PRIVATE "-fborland-exec-code-page=65001")
 adecc_test_use_core(${_target} mssql)
 target_include_directories(${_target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../common")
 target_link_libraries(${_target} PRIVATE odbc32)
 if(_target STREQUAL "adecc_mssql_csv_pipeline")
  adecc_test_use_core(${_target} openxlsx)
  target_link_libraries(${_target} PRIVATE OpenXLSX::OpenXLSX)
  adecc_test_install(${_target} PACKAGE_ROOTS
   "${OPENXLSX_PACKAGE}" "${PUGIXML_PACKAGE}" "${LIBZIP_PACKAGE}" "${BOOST_PACKAGE}"
   TOOLCHAIN_DLLS libc++-370.dll libunwind-370.dll)
 else()
  adecc_test_install(${_target} TOOLCHAIN_DLLS libc++-370.dll libunwind-370.dll)
 endif()
endforeach()
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/bookings.csv" DESTINATION ".")
