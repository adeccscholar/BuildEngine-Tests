cmake_minimum_required(VERSION 3.25)
project(BuildEngineCompareTest LANGUAGES CXX)
get_filename_component(BUILDENGINE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(BUILDENGINE_INSTALL_ROOT "${BUILDENGINE_ROOT}/install" CACHE PATH "BuildEngine install root")
set(BUILDENGINE_CONFIGURATION "Release" CACHE STRING "BuildEngine package variant")
set(_pkg "${BUILDENGINE_INSTALL_ROOT}/packages")
set(_ff "${_pkg}/fast_float/8.2.3")
set(_tbb "${_pkg}/onetbb/2023.1.0")
if(NOT EXISTS "${_ff}/include/fast_float/fast_float.h")
  message(FATAL_ERROR "BuildEngine fast_float 8.2.3 missing")
endif()
find_package(TBB CONFIG REQUIRED COMPONENTS tbb PATHS "${_tbb}/lib/win64/${BUILDENGINE_CONFIGURATION}/cmake/TBB" NO_DEFAULT_PATH)
function(add_compare_executable target file)
  add_executable(${target} ${file} MyProcess.cpp Geodaten.cpp)
  set_property(TARGET ${target} PROPERTY CXX_SCAN_FOR_MODULES OFF)
  target_compile_features(${target} PRIVATE cxx_std_23)
  target_compile_options(${target} PRIVATE -std=c++23)
  target_include_directories(${target} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${_ff}/include")
  target_link_libraries(${target} PRIVATE TBB::tbb)
  target_compile_definitions(${target} PRIVATE __TBB_NO_IMPLICIT_LINKAGE=1)
  install(TARGETS ${target} RUNTIME DESTINATION bin)
endfunction()
add_compare_executable(compare-test Test.cpp)
option(COMPARE_BUILD_TUNE "Build alternative benchmark" OFF)
if(COMPARE_BUILD_TUNE)
  add_compare_executable(compare-tune Test_Tune.cpp)
endif()
